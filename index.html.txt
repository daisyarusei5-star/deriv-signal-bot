/* ============================================================  
   DIGIT EXTRACTION  
============================================================ */  
function getLastDigit(price) {  
  if (price === null || price === undefined) return null;  
  const n = Number(price);  
  if (!Number.isFinite(n)) return null;  
  if (currentPipSize && Number.isFinite(Number(currentPipSize))) {  
    const pip = Number(currentPipSize);  
    if (pip > 0) return Math.abs(Math.round(n / pip) % 10);  
  }  
  const str = String(price);  
  const digits = str.replace(/\D/g, "");  
  if (!digits.length) return null;  
  return Number(digits.charAt(digits.length - 1));  
}  
  
function formatPrice(price) {  
  if (price === null || price === undefined) return "—";  
  const n = Number(price);  
  if (!Number.isFinite(n)) return "—";  
  if (currentPipSize && currentPipSize >= 0.1) {  
    const dec = Math.max(0, Math.round(-Math.log10(currentPipSize)));  
    return n.toLocaleString(undefined, { minimumFractionDigits: dec, maximumFractionDigits: dec });  
  }  
  return n.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 5 });  
}  
  
/* ============================================================  
   DIGIT ANALYSIS  
============================================================ */  
function countDigits(data) {  
  const c = Array(10).fill(0);  
  data.forEach(t => { const d = getLastDigit(t.price); if (d !== null) c[d]++; });  
  return c;  
}  
  
function digitProbabilities(data) {  
  const c = countDigits(data);  
  const total = c.reduce((a, b) => a + b, 0);  
  return c.map(v => (v + 1) / (total + 10));  
}  
  
function weightedDigitProbabilities(data) {  
  const w = Array(10).fill(1);  
  let totalW = 0;  
  const len = data.length;  
  data.forEach((t, i) => {  
    const d = getLastDigit(t.price);  
    if (d === null) return;  
    const age = len - i;  
    const weight = 0.6 + 0.4 * (1 - age / Math.max(len, 1));  
    w[d] += weight;  
    totalW += weight;  
  });  
  const s = 1;  
  return w.map(v => (v + s) / (totalW + s * 10));  
}  
  
function topDigit(data) {  
  const p = weightedDigitProbabilities(data);  
  let idx = 0;  
  for (let i = 1; i < 10; i++) if (p[i] > p[idx]) idx = i;  
  return { digit: idx, probability: p[idx], probabilities: p };  
}  
  
function bottomDigit(data) {  
  const p = weightedDigitProbabilities(data);  
  let idx = 0;  
  for (let i = 1; i < 10; i++) if (p[i] < p[idx]) idx = i;  
  return { digit: idx, probability: p[idx], probabilities: p };  
}  
  
function evenOddProbabilities(data) {  
  const p = weightedDigitProbabilities(data);  
  let even = 0, odd = 0;  
  for (let d = 0; d <= 9; d++) { if (d % 2 === 0) even += p[d]; else odd += p[d]; }  
  return { even, odd };  
}  
  
function overUnderProbabilities(data) {  
  const p = weightedDigitProbabilities(data);  
  let bestOver = { prob: 0, barrier: 0 }, bestUnder = { prob: 0, barrier: 9 };  
  for (let b = 0; b <= 8; b++) {  
    let over = 0;  
    for (let d = b + 1; d <= 9; d++) over += p[d];  
    if (over > bestOver.prob) bestOver = { prob: over, barrier: b };  
  }  
  for (let b = 1; b <= 9; b++) {  
    let under = 0;  
    for (let d = 0; d < b; d++) under += p[d];  
    if (under > bestUnder.prob) bestUnder = { prob: under, barrier: b };  
  }  
  return { over: bestOver, under: bestUnder };  
}  
  
function directionProbability(data) {  
  if (data.length < 2) return { rise: 0.5, fall: 0.5 };  
  let rises = 0, falls = 0;  
  for (let i = 1; i < data.length; i++) {  
    const a = Number(data[i - 1].price), b = Number(data[i].price);  
    if (Number.isFinite(a) && Number.isFinite(b)) { if (b > a) rises++; else if (b < a) falls++; }  
  }  
  const total = rises + falls;  
  if (!total) return { rise: 0.5, fall: 0.5 };  
  return { rise: rises / total, fall: falls / total };  
}  
  
function stability(data) {  
  if (data.length < 10) return 0;  
  const half = Math.floor(data.length / 2);  
  const a = digitProbabilities(data.slice(0, half));  
  const b = digitProbabilities(data.slice(half));  
  let dist = 0;  
  for (let i = 0; i < 10; i++) dist += Math.abs(a[i] - b[i]);  
  return Math.max(0, Math.min(100, 100 - dist * 100));  
}  
  
/* ============================================================  
   MOMENTUM — recent direction strength  
============================================================ */  
function momentum(data) {  
  if (data.length < 10) return 0;  
  const recent = data.slice(-20);  
  let up = 0, down = 0;  
  for (let i = 1; i < recent.length; i++) {  
    const a = Number(recent[i-1].price), b = Number(recent[i].price);  
    if (Number.isFinite(a) && Number.isFinite(b)) {  
      if (b > a) up += (b - a);  
      else if (b < a) down += (a - b);  
    }  
  }  
  const total = up + down;  
  if (!total) return 0;  
  return ((up - down) / total) * 100; // -100 to +100  
}  
  
/* ============================================================  
   VOLATILITY CHECK — is the market behaving normally?  
============================================================ */  
function volatilityCheck(data) {  
  if (data.length < 20) return { valid: false, score: 0 };  
  const digits = data.map(t => getLastDigit(t.price)).filter(d => d !== null);  
  if (digits.length < 20) return { valid: false, score: 0 };  
    
  // Check if digits are reasonably distributed (not all same)  
  const counts = Array(10).fill(0);  
  digits.forEach(d => counts[d]++);  
  const maxCount = Math.max(...counts);  
  const expected = digits.length / 10;  
    
  // If one digit appears more than 3x expected, market may be abnormal  
  if (maxCount > expected * 3) return { valid: false, score: 30 };  
    
  // Calculate how spread out the distribution is  
  const deviations = counts.map(c => Math.abs(c - expected));  
  const avgDeviation = deviations.reduce((a,b) => a+b, 0) / 10;  
  const maxDeviation = Math.max(...deviations);  
    
  // Lower deviation = more predictable = higher score  
  const score = Math.max(0, Math.min(100, 100 - (avgDeviation / expected) * 50));  
  return { valid: true, score };  
}  
  
/* ============================================================  
   STREAKS  
============================================================ */  
function calculateStreaks(data) {  
  if (data.length < 2) return { eoStreak: 0, eoType: null, digitStreak: 0, digitType: null, longEven: 0, longOdd: 0, rfStreak: 0, rfType: null };  
  
  let eoStreak = 1, eoType = null, longEven = 0, longOdd = 0;  
  let digitStreak = 1, digitType = null;  
  let rfStreak = 1, rfType = null;  
  
  const digits = data.map(t => getLastDigit(t.price)).filter(d => d !== null);  
  const dirs = [];  
  for (let i = 1; i < data.length; i++) {  
    const a = Number(data[i - 1].price), b = Number(data[i].price);  
    if (Number.isFinite(a) && Number.isFinite(b)) dirs.push(b > a ? "rise" : b < a ? "fall" : null);  
  }  
  
  if (digits.length >= 2) {  
    eoType = digits[digits.length - 1] % 2 === 0 ? "even" : "odd";  
    for (let i = digits.length - 2; i >= 0; i--) {  
      const cur = digits[i] % 2 === 0 ? "even" : "odd";  
      if (cur === eoType) eoStreak++;  
      else break;  
    }  
  }  
  
  let curEven = 0, curOdd = 0;  
  digits.forEach(d => {  
    if (d % 2 === 0) { curEven++; curOdd = 0; longEven = Math.max(longEven, curEven); }  
    else { curOdd++; curEven = 0; longOdd = Math.max(longOdd, curOdd); }  
  });  
  
  if (digits.length >= 2) {  
    digitType = digits[digits.length - 1];  
    for (let i = digits.length - 2; i >= 0; i--) {  
      if (digits[i] === digitType) digitStreak++;  
      else break;  
    }  
  }  
  
  if (dirs.length >= 2) {  
    rfType = dirs[dirs.length - 1];  
    for (let i = dirs.length - 2; i >= 0; i--) {  
      if (dirs[i] === rfType) rfStreak++;  
      else break;  
    }  
  }  
  
  return { eoStreak, eoType, digitStreak, digitType, longEven, longOdd, rfStreak, rfType };  
}  
  
/* ============================================================  
   SIGNAL ENGINE — STRICT MULTI-FACTOR SCORING  
   Enhanced with momentum, volatility, and stricter thresholds  
============================================================ */  
function computeSignal() {  
  if (ticks.length < 200) {  
    return {  
      state: "wait", score: 0, confidence: 0, edge: 0,  
      target: "NO ENTRY", reason: "Need at least 200 ticks. Currently: " + ticks.length,

    };

  const winSize = Number(windowEl.value);  
  const data = ticks.slice(Math.max(0, ticks.length - winSize));  
  const probs = weightedDigitProbabilities(data);  
  const top = topDigit(data);  
  const bottom = bottomDigit(data);  
  const eo = evenOddProbabilities(data);  
  const ou = overUnderProbabilities(data);  
  const dir = directionProbability(data);  
  const stab = stability(data);  
  const streaks = calculateStreaks(ticks);  
  const mom = momentum(data);  
  const vol = volatilityCheck(data);  
  
  // Volatility check — reject if market is abnormal  
  if (!vol.valid) {  
    return {  
      state: "wait", score: 0, confidence: 0, edge: 0,  
      target: "NO ENTRY",  
      reason: "⚠ Market volatility abnormal — digit distribution too skewed. Wait for normalization.",  
      contract: null, digit: null, quality: vol.score, guide: null  
    };  
  }  
  
  // Multi-window agreement  
  const winData = [50, 100, 500].map(s => ticks.slice(Math.max(0, ticks.length - s)));  
  const winTops = winData.map(d => topDigit(d).digit);  
  const digitAgree = winTops.filter(x => x === winTops[0]).length / winTops.length * 100;  
  
  const winEo = winData.map(d => {  
    const e = evenOddProbabilities(d);  
    return e.even >= e.odd ? "even" : "odd";  
  });  
  const eoAgree = winEo.filter(x => x === winEo[0]).length / winEo.length * 100;  
  
  const winDirs = winData.map(d => {  
    const p = directionProbability(d);  
    return p.rise >= p.fall ? "rise" : "fall";  
  });  
  const dirAgree = winDirs.filter(x => x === winDirs[0]).length / winDirs.length * 100;  
  
  // Evaluate each contract type — STRICTER thresholds  
  const candidates = [];  
  
  // Matches (specific digit) — STRICT  
  const matchesEdge = (top.probability - 0.10) * 100;  
  if (top.probability >= 0.14 && digitAgree >= 80 && stab >= 55) {  
    candidates.push({  
      contract: "Matches",  
      target: "Digit " + top.digit,  
      digit: top.digit,  
      probability: top.probability,  
      edge: matchesEdge,  
      agreement: digitAgree,  
      score: top.probability * 30 + digitAgree * 0.3 + stab * 0.15 + Math.max(0, matchesEdge) * 2,  
      guide: buildMatchesGuide(top.digit, top.probability, streaks)  
    });  
  }  
  
  // Differs (not a specific digit) — STRICT  
  const differsProb = 1 - bottom.probability;  
  const differsEdge = (differsProb - 0.90) * 100;  
  if (differsProb >= 0.93 && stab >= 50) {  
    candidates.push({  
      contract: "Differs",  
      target: "≠ " + bottom.digit,  
      digit: bottom.digit,  
      probability: differsProb,  
      edge: differsEdge,  
      agreement: 100,  
      score: differsProb * 20 + Math.max(0, differsEdge) * 3 + stab * 0.15,  
      guide: buildDiffersGuide(bottom.digit, differsProb, streaks)  
    });  
  }  
  
  // Even — STRICT  
  if (eo.even >= 0.55 && eoAgree >= 80 && stab >= 55) {  
    candidates.push({  
      contract: "Even",  
      target: "EVEN",  
      digit: null,  
      probability: eo.even,  
      edge: (eo.even - 0.50) * 100,  
      agreement: eoAgree,  
      score: eo.even * 25 + eoAgree * 0.3 + stab * 0.15 + Math.max(0, (eo.even - 0.50) * 100) * 2,  
      guide: buildEvenOddGuide("Even", eo.even, streaks)  
    });  
  }  
  
  // Odd — STRICT  
  if (eo.odd >= 0.55 && eoAgree >= 80 && stab >= 55) {  
    candidates.push({  
      contract: "Odd",  
      target: "ODD",  
      digit: null,  
      probability: eo.odd,  
      edge: (eo.odd - 0.50) * 100,  
      agreement: eoAgree,  
      score: eo.odd * 25 + eoAgree * 0.3 + stab * 0.15 + Math.max(0, (eo.odd - 0.50) * 100) * 2,  
      guide: buildEvenOddGuide("Odd", eo.odd, streaks)  
    });  
  }  
  
  // Over — STRICT  
  if (ou.over.prob >= 0.58 && stab >= 50) {  
    candidates.push({  
      contract: "Over",  
      target: "Over " + ou.over.barrier,  
      digit: ou.over.barrier,  
      probability: ou.over.prob,  
      edge: (ou.over.prob - 0.50) * 100,  
      agreement: digitAgree,  
      score: ou.over.prob * 22 + Math.max(0, (ou.over.prob - 0.50) * 100) * 2.5 + stab * 0.1,  
      guide: buildOverUnderGuide("Over", ou.over.barrier, ou.over.prob, streaks)  
    });  
  }  
  
  // Under — STRICT  
  if (ou.under.prob >= 0.58 && stab >= 50) {  
    candidates.push({  
      contract: "Under",  
      target: "Under " + ou.under.barrier,  
      digit: ou.under.barrier,  
      probability: ou.under.prob,  
      edge: (ou.under.prob - 0.50) * 100,  
      agreement: digitAgree,  
      score: ou.under.prob * 22 + Math.max(0, (ou.under.prob - 0.50) * 100) * 2.5 + stab * 0.1,  
      guide: buildOverUnderGuide("Under", ou.under.barrier, ou.under.prob, streaks)  
    });  
  }  
  
  // Rise — STRICT (needs momentum confirmation)  
  if (dir.rise >= 0.58 && dirAgree >= 80 && mom > 5) {  
    candidates.push({  
      contract: "Rise",  
      target: "RISE",  
      digit: null,  
      probability: dir.rise,  
      edge: (dir.rise - 0.50) * 100,  
      agreement: dirAgree,  
      score: dir.rise * 25 + dirAgree * 0.3 + Math.max(0, mom) * 0.2 + Math.max(0, (dir.rise - 0.50) * 100) * 2,  
      guide: buildRiseFallGuide("Rise", dir.rise, mom, streaks)  
    });

  // Fall — STRICT (needs momentum confirmation)

    candidates.push({  
      contract: "Fall",  
      target: "FALL",  
      digit: null,  
      probability: dir.fall,  
      edge: (dir.fall - 0.50) * 100,  
      agreement: dirAgree,  
      score: dir.fall * 25 + dirAgree * 0.3 + Math.max(0, -mom) * 0.2 + Math.max(0, (dir.fall - 0.50) * 100) * 2,  
      guide: buildRiseFallGuide("Fall", dir.fall, mom, streaks)  
    });  
  }  
  
  // Streak reversal bonus  
  if (streaks.eoStreak >= 4) {  
    const reversalType = streaks.eoType === "even" ? "Odd" : "Even";  
    const reversalProb = streaks.eoType === "even" ? eo.odd : eo.even;  
    const idx = candidates.findIndex(c => c.contract === reversalType);

      candidates[idx].score += streaks.eoStreak * 5;  
      candidates[idx].guide = buildStreakReversalGuide(reversalType, streaks.eoStreak, streaks.eoType, reversalProb);  
    }  
  }  
  
  // Sort by score  
  candidates.sort((a, b) => b.score - a.score);  
  
  // STRICT minimum score threshold  
  if (candidates.length === 0 || candidates[0].score < 45) {  
    let reason = "No contract meets the strict quality threshold. ";  
    if (candidates.length > 0) reason += "Best candidate: " + candidates[0].contract + " (" + Math.round(candidates[0].score) + "% — needs 45%+)";  
    else reason += "Waiting for stronger alignment across all filters.";  
    return {  
      state: "wait",  
      score: candidates.length > 0 ? Math.round(candidates[0].score) : 0,  
      confidence: 0, edge: 0,  
      target: "NO ENTRY",  
      reason,  
      contract: null, digit: null, quality: vol.score, guide: null  
    };  
  }  
  
  const best = candidates[0];  
  const confidence = Math.min(100, Math.round(best.score));  
  const quality = Math.round(  
    Math.min(100,  
      Math.min(20, ticks.length / 500 * 20) +  
      best.agreement * 0.3 +  
      Math.max(0, Math.min(25, best.edge * 3)) +  
      stab * 0.2 +  
      vol.score * 0.1  
    )  
  );  
  
  // STRICT: need both confidence AND quality high  
  let state = "wait";  
  if (confidence >= 75 && quality >= 70) state = "good";  
  
  let reason = best.contract + " signal: " + (best.probability * 100).toFixed(1) + "% probability";  
  reason += " • Agreement: " + best.agreement.toFixed(0) + "%";  
  reason += " • Stability: " + stab.toFixed(0) + "%";  
  if (best.contract === "Rise" || best.contract === "Fall") reason += " • Momentum: " + mom.toFixed(1);  
  
  return {  
    state, score: confidence, confidence, edge: best.edge,  
    target: best.target, reason,  
    contract: best.contract, digit: best.digit, quality, guide: best.guide  
  };  
}

function render() {  
  $("ticksLoaded").textContent = ticks.length;  
  if (ticks.length) {  
    const latest = ticks[ticks.length - 1];  
    $("lastDigit").textContent = getLastDigit(latest.price);  
    $("lastPrice").textContent = formatPrice(latest.price);  
  } else {  
    $("lastDigit").textContent = "—";  
    $("lastPrice").textContent = "—";  
  }  
  
  const sig = computeSignal();  
  signalEl.className = "signal " + sig.state;  
  signalMain.textContent = sig.state === "good" ? "GOOD ENTRY" : "WAIT";  
  signalSub.textContent = sig.reason;  
  signalTarget.textContent = sig.target;  
  
  $("quality").textContent = sig.quality ? sig.quality + "%" : "—";  
  $("quality").className = "metricValue " + (sig.quality >= 70 ? "green" : sig.quality >= 50 ? "yellow" : "");  
  $("confidence").textContent = sig.confidence ? sig.confidence + "%" : "—";  
  $("confidence").className = "metricValue " + (sig.confidence >= 75 ? "green" : sig.confidence >= 50 ? "yellow" : "");  
  $("edge").textContent = ticks.length >= 30 ? ((sig.edge >= 0 ? "+" : "") + sig.edge.toFixed(1) + "%") : "—";  
  $("edge").className = "metricValue " + (sig.edge >= 3 ? "green" : sig.edge > 0 ? "yellow" : sig.edge < 0 ? "red" : "");  
  
  // Show/hide guide  
  if (sig.state === "good" && sig.guide) {  
    guideEl.style.display = "block";  
    guideSteps.innerHTML = "";  
    sig.guide.forEach((step, i) => {  
      const div = document.createElement("div");  
      div.className = "guideStep";  
      div.innerHTML = '<div class="guideNum">' + (i + 1) + '</div><div class="guideText">' + step.text + '</div>';  
      guideSteps.appendChild(div);  
    });  
  } else {  
    guideEl.style.display = "none";  
  }  
  
  // Track signal state changes for sound + history  
  if (sig.state === "good" && lastSignalState !== "good") {  
    playAlert("good");  
    addHistory(sig);  
    // Set target digit for tick alert  
    if (sig.digit !== null && sig.digit !== undefined) {  
      targetDigit = sig.digit;  
      targetContract = sig.contract;  
    }  
  }  
  lastSignalState = sig.state;  
  
  // Tick alert — if we have a good entry and the target digit appears  
  if (sig.state === "good" && sig.digit !== null && ticks.length) {  
    const lastD = getLastDigit(ticks[ticks.length - 1].price);  
    if (lastD === sig.digit && sig.contract !== "Matches" && sig.contract !== "Differs") {  
      // For Over/Under, alert when digit touches the barrier  
      playAlert("tick");  
    }  
  }  
  
  // Digit distribution  
  const winSize = Number(windowEl.value);  
  const data = ticks.slice(Math.max(0, ticks.length - winSize));  
  const probs = weightedDigitProbabilities(data);  
  const sorted = probs.map((p, d) => ({ d, p })).sort((a, b) => b.p - a.p);  
  const hot = sorted[0] ? sorted[0].d : null;  
  const cold = sorted[sorted.length - 1] ? sorted[sorted.length - 1].d : null;  
  const targetD = sig.state === "good" ? sig.digit : null;  
  digitGrid.innerHTML = "";  
  for (let d = 0; d < 10; d++) {  
    const pct = probs[d] * 100;  
    const box = document.createElement("div");  
    let cls = "digit";  
    if (d === hot) cls += " hot";  
    if (d === cold) cls += " cold";  
    if (d === targetD) cls += " target";  
    box.className = cls;  
    let tag = d === hot ? "HOT" : d === cold ? "COLD" : d === targetD ? "TARGET" : "";  
    box.innerHTML = '<div class="digitNumber">' + d + '</div><div class="digitPct">' + pct.toFixed(1) + '%</div><div class="digitTag">' + tag + '</div>';  
    digitGrid.appendChild(box);  
  }  
  
  // Even / Odd  
  const eo = evenOddProbabilities(data);  
  $("evenPct").textContent = (eo.even * 100).toFixed(1) + "%";  
  $("oddPct").textContent = (eo.odd * 100).toFixed(1) + "%";  
  $("evenBar").style.width = (eo.even * 100) + "%";  
  $("oddBar").style.width = (eo.odd * 100) + "%";  
  
  // Over / Under  
  const ou = overUnderProbabilities(data);  
  $("overPct").textContent = (ou.over.prob * 100).toFixed(1) + "%";  
  $("overTarget").textContent = "Barrier: >" + ou.over.barrier;  
  $("overBar").style.width = (ou.over.prob * 100) + "%";  
  $("underPct").textContent = (ou.under.prob * 100).toFixed(1) + "%";  
  $("underTarget").textContent = "Barrier: <" + ou.under.barrier;  
  $("underBar").style.width = (ou.under.prob * 100) + "%";  
  
  // Rise / Fall  
  const dir = directionProbability(data);  
  $("risePct").textContent = (dir.rise * 100).toFixed(1) + "%";  
  $("fallPct").textContent = (dir.fall * 100).toFixed(1) + "%";  
  $("riseBar").style.width = (dir.rise * 100) + "%";  
  $("fallBar").style.width = (dir.fall * 100) + "%";  
  
  // Multi-window  
  [50, 100, 500].forEach(s => {  
    const wd = ticks.slice(Math.max(0, ticks.length - s));  
    if (wd.length < 5) {  
      $("top" + s).textContent = "—";  
      $("prob" + s).textContent = "—";  
      $("agree" + s).textContent = "—";  
    } else {  
      const t = topDigit(wd);  
      $("top" + s).textContent = t.digit;  
      $("prob" + s).textContent = (t.probability * 100).toFixed(1) + "%";  
      const allTops = [50, 100, 500].map(x => topDigit(ticks.slice(Math.max(0, ticks.length - x))).digit);  
      const match = allTops.filter(x => x === t.digit).length;  
      $("agree" + s).textContent = match + "/3 agree";  
    }  
  });  
  
  // Streaks  
  const streaks = calculateStreaks(ticks);  
  $("eoStreak").textContent = streaks.eoStreak > 0 ? streaks.eoStreak + "x " + (streaks.eoType || "").toUpperCase() : "—";  
  $("eoStreak").className = "streakValue " + (streaks.eoStreak >= 4 ? (streaks.eoType === "even" ? "green" : "red") : "");  
  $("digitStreak").textContent = streaks.digitStreak > 0 ? streaks.digitStreak + "x digit " + streaks.digitType : "—";  
  $("longEven").textContent = streaks.longEven > 0 ? streaks.longEven + "x" : "—";  
  $("longOdd").textContent = streaks.longOdd > 0 ? streaks.longOdd + "x" : "—";  
  $("rfStreak").textContent = streaks.rfStreak > 0 ? streaks.rfStreak + "x " + (streaks.rfType || "").toUpperCase() : "—";  
  
  // Recent ticks  
  tickList.innerHTML = "";  
  const recent = ticks.slice(-35).reverse();  
  if (!recent.length) {  
    tickList.innerHTML = '<span style="color:#5d6c79;font-size:10px">No ticks yet</span>';  
  } else {  
    recent.forEach(t => {  
      const d = getLastDigit(t.price);  
      const chip = document.createElement("div");  
      let cls = "tickChip " + (d % 2 === 0 ? "even" : "odd");  
      // Highlight if this is the target digit for current signal  
      if (sig.state === "good" && d === sig.digit && sig.digit !== null) cls += " alert";  
      chip.className = cls;  
      chip.textContent = d === null ? "?" : d;  
      chip.title = formatPrice(t.price);  
      tickList.appendChild(chip);  
    });  
  }  
}

function addHistory(sig) {  
  const time = new Date().toLocaleTimeString();  
  signalHistory.unshift({ time, state: sig.state, target: sig.target, score: sig.confidence, contract: sig.contract });  
  if (signalHistory.length > 10) signalHistory.pop();  
  renderHistory();  
}  
  
function renderHistory() {  
  if (!signalHistory.length) {  
    historyList.innerHTML = '<span style="color:#5d6c79;font-size:10px">No signals yet</span>';  
    return;  
  }  
  historyList.innerHTML = "";  
  signalHistory.forEach(h => {  
    const item = document.createElement("div");  
    item.className = "histItem";  
    const badgeClass = h.state === "good" ? "good" : h.state === "wait" ? "wait" : "danger";  
    const icon = h.state === "good" ? "✓" : h.state === "wait" ? "○" : "✗";  
    item.innerHTML = '<div class="histLeft"><div class="histBadge ' + badgeClass + '">' + icon + '</div><div class="histInfo"><strong>' + (h.contract || "—") + '</strong><span>' + h.time + ' • ' + h.target + '</span></div></div><div class="histScore ' + badgeClass + '">' + h.score + '%</div>';  
    historyList.appendChild(item);  
  });  
}  
  
/* ============================================================  
   WEBSOCKET  
============================================================ */  
function connect() {  
  if (!running) return;  
  clearTimeout(reconnectTimer);  
  statusEl.textContent = "Connecting to Deriv live feed...";  
  try { socket = new WebSocket(DERIV_WS); } catch (e) { scheduleReconnect(); return; }  
  
  socket.addEventListener("open", () => {  
    statusEl.textContent = "Connected. Loading historical ticks...";  
    requestHistory();  
  });  
  
  socket.addEventListener("message", event => {  
    try { handleMessage(JSON.parse(event.data)); } catch (e) {}  
  });  
  
  socket.addEventListener("close", () => {  
    if (running) { statusEl.textContent = "Connection lost. Reconnecting..."; scheduleReconnect(); }  
  });  
  
  socket.addEventListener("error", () => { statusEl.textContent = "Connection error."; });  
}  
  
function requestHistory() {  
  if (!socket || socket.readyState !== WebSocket.OPEN) return;  
  socket.send(JSON.stringify({ ticks_history: marketEl.value, count: 1000, end: "latest", style: "ticks", req_id: 1 }));  
}  
  
function subscribeTicks() {  
  if (!socket || socket.readyState !== WebSocket.OPEN) return;  
  socket.send(JSON.stringify({ ticks: marketEl.value, subscribe: 1, req_id: 2 }));  
}  
  
function handleMessage(data) {  
  if (data.pip_size) currentPipSize = Number(data.pip_size);  
  if (data.echo_req && data.echo_req.pip_size) currentPipSize = Number(data.echo_req.pip_size);  
  
  if (data.history && Array.isArray(data.history.prices)) {  
    ticks = data.history.prices  
      .map((price, i) => ({ price, epoch: Array.isArray(data.history.times) ? data.history.times[i] : Date.now() / 1000 }))  
      .filter(x => Number.isFinite(Number(x.price)));  
    ticks = ticks.slice(-2000);  
    statusEl.textContent = "Historical data loaded. Live stream active.";  
    render();  
    subscribeTicks();  
    return;  
  }  
  
  if (data.tick) {  
    const tick = data.tick;  
    if (tick.pip_size) currentPipSize = Number(tick.pip_size);  
    const item = { price: tick.quote, epoch: tick.epoch || Date.now() / 1000 };  
    if (item.price !== undefined && item.price !== null) {  
      ticks.push(item);  
      if (ticks.length > 2000) ticks = ticks.slice(-2000);  
      statusEl.textContent = "Live • " + ticks.length + " ticks";  
      render();  
    }  
  }  
}  
  
function scheduleReconnect() {  
  clearTimeout(reconnectTimer);  
  reconnectTimer = setTimeout(connect, 2500);  
}  
  
/* ============================================================  
   START / STOP  
============================================================ */  
function startAnalysis() {  
  if (running) { stopAnalysis(); return; }  
  running = true;  
  startBtn.classList.add("stop");  
  startBtn.textContent = "■ STOP";  
  statusEl.textContent = "Starting...";  
  ticks = [];  
  currentPipSize = null;  
  signalHistory = [];  
  lastSignalState = null;  
  renderHistory();  
  render();  
  connect();  
}  
  
function stopAnalysis() {  
  running = false;  
  clearTimeout(reconnectTimer);  
  if (socket) { try { socket.close(); } catch (e) {} socket = null; }  
  startBtn.classList.remove("stop");  
  startBtn.textContent = "▶ START ANALYSIS";  
  statusEl.textContent = "Stopped. Press START to begin.";  
}  
  
marketEl.addEventListener("change", () => {  
  ticks = []; currentPipSize = null; signalHistory = []; lastSignalState = null; renderHistory(); render();  
  if (running) { if (socket) { try { socket.close(); } catch (e) {} } setTimeout(connect, 300); }  
});  
  
windowEl.addEventListener("change", render);  
startBtn.addEventListener("click", startAnalysis);  
  
/* ============================================================

render();  
// Auto-start after a short delay (allows audio context to be unlocked on user interaction)  
setTimeout(() => {  
  startAnalysis();
